1. Tela "Carregamento" com lista de pedidos de TODAS as contas simultaneamente, com badge indicando qual conta (RP/TI/VO)

7. Opcao de imprimir a lista consolidada de carregamento na impressora termica
8. Produtos com mesmo nome+variante de contas diferentes sao SOMADOS na lista consolidada
9. ActionBar fixa no bottom com resumo recalculado em tempo real ao marcar/desmarcar
10. BottomSheet com lista consolidada completa, resumo financeiro e pedidos incluidos

## Dev Notes

### CONTEXTO: POR QUE ESTA E A DOR PRINCIPAL

Hoje o Tiago opera com 3 celulares, cada um com o app Kite numa conta diferente (Distrial RP, Tiago, Votoporanga). Para carregar a van, ele precisa:
1. Abrir o Kite no celular 1, somar os produtos dos pedidos de RP
2. Abrir o Kite no celular 2, somar os produtos dos pedidos de Tiago
3. Abrir o Kite no celular 3, somar os produtos dos pedidos de Votoporanga
4. Somar tudo manualmente num papel

Isso leva ~15 minutos por dia e e propenso a erros. Esta tela resolve isso em <30 segundos: seleciona os pedidos de todas as contas, ve a lista consolidada automatica, imprime e vai.

### DATABASE: Como Funciona Cross-Account

A chave e que o RLS do Supabase usa `get_user_account_ids()` que retorna TODAS as contas do usuario. Quando Tiago esta logado, a funcao retorna os IDs das 3 contas. Portanto, uma query em `orders` sem filtro de `account_id` especifico ja retorna pedidos de todas as contas:

```sql
-- RLS policy em orders (ja criada no Epic 1):
CREATE POLICY "Users can view own account orders"
  ON public.orders FOR SELECT
  USING (account_id = ANY(get_user_account_ids()));

-- Essa query ja retorna pedidos cross-account automaticamente:
SELECT o.*, c.name as client_name, a.name as account_name, a.slug as account_slug
FROM orders o
JOIN clients c ON o.client_id = c.id
JOIN accounts a ON o.account_id = a.id
WHERE o.status IN ('lancado', 'confirmado')
ORDER BY o.created_at DESC;
```

### SQL: Consolidacao de Produtos (Server-Side Backup)

Se necessario fazer server-side (para impressao ou API), a query de consolidacao e:

```sql
-- Consolida produtos de pedidos selecionados (cross-account)
SELECT
  oi.product_name,
  oi.variant_name,
  SUM(oi.quantity) as total_quantity,
  -- Para referencia: detalhes por conta
  json_agg(json_build_object(
    'account_name', a.name,
    'account_slug', a.slug,
    'quantity', oi.quantity,
    'order_id', oi.order_id
  )) as breakdown
FROM order_items oi
JOIN orders o ON oi.order_id = o.id
JOIN accounts a ON o.account_id = a.id
WHERE o.id = ANY($1::uuid[])  -- array de order_ids selecionados
  AND oi.is_returned = false
GROUP BY oi.product_name, oi.variant_name
ORDER BY total_quantity DESC;

-- Totais financeiros
SELECT
  SUM(o.total) as total_revenue,
  SUM(o.total_cost) as total_cost,
  SUM(o.profit) as total_profit,
  COUNT(o.id) as order_count,
  COUNT(DISTINCT o.account_id) as account_count
FROM orders o
WHERE o.id = ANY($1::uuid[]);

-- Breakdown por conta
SELECT
  a.name as account_name,
  a.slug as account_slug,
  SUM(o.total) as account_revenue,
  COUNT(o.id) as account_order_count
FROM orders o
JOIN accounts a ON o.account_id = a.id
WHERE o.id = ANY($1::uuid[])
GROUP BY a.name, a.slug
ORDER BY account_revenue DESC;
```

### API Route: POST /api/loading/consolidate

// apps/web/src/app/api/loading/consolidate/route.ts
import { createRouteHandlerClient } from '@supabase/auth-helpers-nextjs';

export async function POST(request: Request) {
  const { order_ids } = await request.json();
  const supabase = createRouteHandlerClient({ cookies });

  // 1. Fetch order items (RLS protects access)
  const { data: items } = await supabase
    .from('order_items')
    .select(`
      product_name, variant_name, quantity,
      order:orders!inner(id, total, total_cost, profit, account_id,
        account:accounts(name, slug),
        client:clients(name)
      )
    `)
    .in('order_id', order_ids)
    .eq('is_returned', false);

  // 2. Consolidate products
  const productMap = new Map<string, { name: string; variant: string; total_quantity: number }>();
  for (const item of items) {
    const key = `${item.product_name}|${item.variant_name}`;
    const existing = productMap.get(key);
    if (existing) {
      existing.total_quantity += item.quantity;
    } else {
      productMap.set(key, {
        name: item.product_name,
        variant: item.variant_name,
        total_quantity: item.quantity,
      });
    }
  }

  // 3. Calculate totals and group by account
  // ... (aggregate from orders)

  return Response.json({
    products: Array.from(productMap.values()).sort((a, b) => b.total_quantity - a.total_quantity),
    total_revenue,
    total_cost,
    total_profit,
    orders_by_account,
  });
}

### CLIENT-SIDE: Logica de Consolidacao (Tempo Real)

```typescript
// apps/web/src/lib/utils/loading-consolidation.ts

export interface ConsolidatedProduct {
  productName: string;
  variantName: string;
  totalQuantity: number;
  unit: string;
}

export interface LoadingSummary {
  totalRevenue: number;
  totalCost: number;
  totalProfit: number;
  orderCount: number;
  accountCount: number;
  accountBreakdown: { accountName: string; slug: string; revenue: number; orderCount: number }[];
}

export function consolidateProducts(orders: OrderWithItems[]): ConsolidatedProduct[] {
  const map = new Map<string, ConsolidatedProduct>();

  for (const order of orders) {
    for (const item of order.items) {
      if (item.is_returned) continue;
      const key = `${item.product_name}|${item.variant_name}`;
      const existing = map.get(key);
      if (existing) {
        existing.totalQuantity += Number(item.quantity) - Number(item.returned_quantity || 0);
      } else {
        map.set(key, {
          productName: item.product_name,
          variantName: item.variant_name,
          totalQuantity: Number(item.quantity) - Number(item.returned_quantity || 0),
          unit: 'un', // default
        });
      }
    }
  }

  return Array.from(map.values())
    .filter(p => p.totalQuantity > 0)
    .sort((a, b) => b.totalQuantity - a.totalQuantity); // maior quantidade primeiro
}

export function calculateLoadingSummary(orders: OrderWithItems[]): LoadingSummary {
  const accountMap = new Map<string, { name: string; slug: string; revenue: number; count: number }>();

  let totalRevenue = 0;
  let totalCost = 0;

  for (const order of orders) {
    totalRevenue += Number(order.total);
    totalCost += Number(order.total_cost);

    const key = order.account_id;
    const existing = accountMap.get(key);
    if (existing) {
      existing.revenue += Number(order.total);
      existing.count++;
    } else {
      accountMap.set(key, {
        name: order.account_name,
        slug: order.account_slug,
        revenue: Number(order.total),
        count: 1,
      });
    }
  }

  return {
    totalRevenue,
    totalCost,
    totalProfit: totalRevenue - totalCost,
    orderCount: orders.length,
    accountCount: accountMap.size,
    accountBreakdown: Array.from(accountMap.values()).map(a => ({
      accountName: a.name,
      slug: a.slug,
      revenue: a.revenue,
      orderCount: a.count,
    })),
  };
}
```

### ZUSTAND LOADING STORE

```typescript
// apps/web/src/stores/loading-store.ts
import { create } from 'zustand';
import { consolidateProducts, calculateLoadingSummary } from '@/lib/utils/loading-consolidation';

interface LoadingState {
  orders: OrderWithItems[];
  selectedIds: Set<string>;
  consolidatedProducts: ConsolidatedProduct[];
  summary: LoadingSummary | null;

  setOrders: (orders: OrderWithItems[]) => void;
  toggleOrder: (id: string) => void;
  selectAll: () => void;
  deselectAll: () => void;
  selectByAccount: (accountId: string) => void;
}

export const useLoadingStore = create<LoadingState>((set, get) => ({
  orders: [],
  selectedIds: new Set(),
  consolidatedProducts: [],
  summary: null,

  setOrders: (orders) => set({ orders }),

  toggleOrder: (id) => {
    const { selectedIds, orders } = get();
    const next = new Set(selectedIds);
    if (next.has(id)) next.delete(id); else next.add(id);

    const selectedOrders = orders.filter(o => next.has(o.id));
    set({
      selectedIds: next,
      consolidatedProducts: consolidateProducts(selectedOrders),
      summary: selectedOrders.length > 0 ? calculateLoadingSummary(selectedOrders) : null,
    });
  },

  selectAll: () => {
    const { orders } = get();
    const allIds = new Set(orders.map(o => o.id));
    set({
      selectedIds: allIds,
      consolidatedProducts: consolidateProducts(orders),
      summary: calculateLoadingSummary(orders),
    });
  },

  deselectAll: () => set({ selectedIds: new Set(), consolidatedProducts: [], summary: null }),

  selectByAccount: (accountId) => {
    const { orders } = get();
    const filtered = orders.filter(o => o.account_id === accountId);
    const ids = new Set(filtered.map(o => o.id));
    set({
      selectedIds: ids,
      consolidatedProducts: consolidateProducts(filtered),
      summary: calculateLoadingSummary(filtered),
    });
  },
}));
```

// apps/web/src/hooks/use-loading.ts

export function useLoadingOrders() {
  return useQuery({
    queryKey: ['loading-orders'],
    queryFn: async () => {
      // RLS automaticamente retorna pedidos de TODAS as contas do usuario
      const { data, error } = await supabase
        .from('orders')
        .select(`
          *,
          client:clients(name, address),
          account:accounts(name, slug),
          items:order_items(*)
        `)
        .in('status', ['lancado', 'confirmado'])
        .order('created_at', { ascending: false });

      if (error) throw error;
      return data;
    },
  });
}
