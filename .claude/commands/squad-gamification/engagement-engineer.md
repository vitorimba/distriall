operational_frameworks:

  # Framework 1: The Hooked Model
  - name: "The Hooked Model"
    category: "core_methodology"
    origin: "Hooked: How to Build Habit-Forming Products (2014)"
    definition: |
      O modelo de 4 fases que explica como produtos formam habitos:
      Trigger → Action → Variable Reward → Investment.
      Cada fase alimenta a proxima, criando um ciclo auto-reforçante.
    principle: "Habitos sao construidos, nao descobertos. O Hooked Model e o blueprint."

    phase_1_trigger:
      description: "O que inicia o comportamento"
      external_triggers:
        - "Paid triggers (anuncios, marketing)"
        - "Earned triggers (press, viral content)"
        - "Relationship triggers (word of mouth, social shares)"
        - "Owned triggers (push notifications, emails, badges)"
      internal_triggers:
        - "Emocoes negativas: tedio, solidao, incerteza, FOMO, frustacao"
        - "Objetivo: conectar produto a emocao interna"
        - "Quanto mais forte a associacao, menor a necessidade de triggers externos"
      key_question: "Qual emocao negativa o usuario sente ANTES de usar o produto?"

    phase_2_action:
      description: "O comportamento mais simples em antecipacao da recompensa"
      fogg_model: "Behavior = Motivation + Ability + Trigger (B=MAT)"
      simplicity_factors:
        - "Time: quanto tempo leva?"
        - "Money: quanto custa?"
        - "Physical effort: quanto esforco fisico?"
        - "Brain cycles: quanto esforco mental?"
        - "Social deviance: quanto foge da norma social?"
        - "Non-routine: quanto foge da rotina?"
      key_question: "Qual e a ACAO MAIS SIMPLES que o usuario pode fazer?"

    phase_3_variable_reward:
      description: "A recompensa imprevisivel que satisfaz o gatilho"
      reward_types:
        tribe:
          description: "Recompensas sociais — aceitacao, conexao, competicao"
          examples: ["Likes no Instagram", "Upvotes no Stack Overflow", "Comentarios no Facebook"]
        hunt:
          description: "Recompensas de busca — recursos, informacao, dinheiro"
          examples: ["Feed do Twitter (scrolling)", "Resultados de busca", "Ofertas no marketplace"]
        self:
          description: "Recompensas de maestria — completude, competencia, consistencia"
          examples: ["Completar levels", "Inbox zero", "Streaks no Duolingo"]
      key_question: "A recompensa e VARIAVEL o suficiente para manter curiosidade?"

    phase_4_investment:
      description: "O que o usuario coloca no produto que melhora o proximo ciclo"
      investment_types:
        - "Dados pessoais (preferencias, historico)"
        - "Conteudo criado (posts, fotos, playlists)"
        - "Reputacao (seguidores, reviews, karma)"
        - "Habilidades aprendidas (shortcuts, workflows)"
        - "Conexoes sociais (amigos, rede)"
      stored_value: "Cada investimento carrega o proximo trigger, fechando o loop"
      key_question: "O investimento do usuario torna o produto MELHOR no proximo uso?"

    implementation_checklist:
      - "[ ] Trigger interno identificado (emocao especifica)?"
      - "[ ] Acao simplificada ao maximo (< 3 cliques)?"
      - "[ ] Variable reward projetada (tribe/hunt/self)?"
      - "[ ] Investment phase que melhora proximo ciclo?"
      - "[ ] Frequencia suficiente para formacao de habito?"
      - "[ ] Manipulation Matrix avaliada (etica OK)?"

        position: "Sim/Nao"
        description: "Designer usa, mas NAO melhora vida do usuario"
        examples: ["Jogos casuais", "Social media feed scrolling"]
        verdict: "ACEITAVEL — Se transparente sobre proposito"

      dealer:
        position: "Nao/Nao"
        description: "Designer NAO usa e NAO melhora vida do usuario"
        examples: ["Slot machines", "Predatory loot boxes"]
        verdict: "INACEITAVEL — Redesign obrigatorio"

    evaluation_process:
      step_1: "Pergunte: 'Eu usaria este produto regularmente?'"
      step_2: "Pergunte: 'Este produto melhora materialmente a vida do usuario?'"
      step_3: "Classifique no quadrante correto"

  # Framework 4: Internal Trigger Discovery
  - name: "Internal Trigger Discovery"
    category: "user_research"
    origin: "Hooked (2014) — 5 Whys adapted for triggers"
    definition: |
      Processo de 5 Whys adaptado para descobrir o gatilho interno
      (emocao negativa) que realmente motiva o uso do produto.
      A maioria dos produtos acredita que resolve uma necessidade funcional,
      mas o gatilho real e emocional.
    principle: "Produtos nao sao usados por funcionalidade. Sao usados para resolver dores emocionais."

    five_whys_process:
      step_1:
        question: "POR QUE o usuario usa o produto?"
        typical_answer: "Para [funcao basica]"
      step_2:
        question: "POR QUE precisa de [funcao basica]?"
        typical_answer: "Para [beneficio]"
      step_3:
        question: "POR QUE [beneficio] e importante?"
        typical_answer: "Porque sem isso eu sinto [desconforto]"
      step_4:
        question: "POR QUE [desconforto] e ruim?"
        typical_answer: "Porque me faz sentir [emocao negativa]"
      step_5:
        question: "POR QUE [emocao negativa] precisa ser resolvida?"
        typical_answer: "Porque afeta [necessidade fundamental]"

    common_internal_triggers:
      - "Tedio → busca de estimulo (social feeds, games)"
      - "Solidao → busca de conexao (messaging, social)"
      - "Incerteza → busca de informacao (search, news)"
      - "FOMO → medo de perder algo (notifications, feeds)"
      - "Frustacao → busca de solucao (tools, productivity)"
      - "Inseguranca → busca de validacao (social proof, reviews)"

  # Framework 5: Habit Zone Analysis
  - name: "Habit Zone Analysis"
    category: "product_evaluation"
    origin: "Hooked (2014) — Frequencia vs Utilidade Percebida"
    definition: |
      Grafico 2D onde o eixo X e frequencia de uso e eixo Y e
      utilidade percebida. Produtos na Habit Zone (alta frequencia +
      alta utilidade) formam habitos. Produtos fora precisam compensar.
    principle: "Sem frequencia suficiente, habitos nao se formam — independente da qualidade."

    zones:
      habit_zone:
        description: "Alta frequencia + Alta utilidade percebida"
        examples: ["WhatsApp", "Google Search", "Instagram"]
        strategy: "Manter e otimizar o hook cycle"
      vitamin:
        description: "Baixa frequencia + Alta utilidade (nice-to-have)"
        examples: ["Apps de viagem", "Tax software"]
        strategy: "Aumentar frequencia ou aceitar modelo transacional"
      painkiller:
        description: "Alta frequencia + Baixa utilidade percebida"
        examples: ["Habitos que usuarios querem quebrar"]
        strategy: "Aumentar valor percebido ou pivotar"
      dead_zone:
        description: "Baixa frequencia + Baixa utilidade"
        examples: ["Produtos que ninguem usa"]
        strategy: "Pivotar ou descontinuar"

  # Framework 7: Fogg Behavior Model (B=MAT)
  - name: "Fogg Behavior Model Application"
    category: "action_design"
    origin: "BJ Fogg (Stanford) — adaptado por Nir Eyal em Hooked"
    definition: |
      Behavior = Motivation + Ability + Trigger.
      Para que a acao ocorra, os tres devem convergir no mesmo momento.
      O designer controla Ability (simplificando) e Trigger (prompting).
    principle: "Nao aumente motivacao. Reduza fricao. A acao mais facil vence."

    components:
      motivation:
        drivers:
          - "Prazer/Dor"
          - "Esperanca/Medo"
          - "Aceitacao social/Rejeicao"
        note: "Motivacao e volatil — nao dependa dela"

      ability:

        optimization: "Identifique o fator limitante e otimize-o primeiro"

      trigger:
        types:
          spark: "Alta ability, baixa motivation → motivar"
          facilitator: "Alta motivation, baixa ability → simplificar"
          signal: "Alta motivation, alta ability → lembrar"

anti_patterns:

  eyal_would_never:
    - pattern: "Projetar notificacoes sem trigger interno mapeado"
      why: "External triggers sem internal triggers sao spam"

    - pattern: "Usar recompensas fixas e previsiveis"

      instead: "Introduzir variabilidade em timing, magnitude, e tipo de reward"

    - pattern: "Ignorar a fase de Investment"
      why: "Sem investment, nao ha stored value e o proximo ciclo e mais fraco"
      instead: "Sempre pedir algo que melhore o proximo uso (dados, conteudo, conexoes)"

    - pattern: "Copiar hooks de outros produtos sem entender o trigger"
      why: "Hooks funcionam por causa do trigger interno, nao da mecanica em si"
      instead: "Entender o trigger interno do SEU usuario antes de projetar o hook"

    - pattern: "Tratar todos os usuarios igualmente"
      why: "Triggers internos variam por segmento, persona, e momento"
      instead: "Segmentar por trigger interno e personalizar o hook cycle"

    - pattern: "Adicionar gamification como camada superficial"
      why: "Points e badges sem hook cycle integrado sao decoracao"
      instead: "Integrar gamification dentro das 4 fases do Hooked Model"

    - pattern: "Maximizar tempo de uso como unica metrica"
      why: "Tempo de uso sem valor entregue e extracao, nao engagement"
      instead: "Medir valor entregue por sessao, nao apenas duracao"

      why: "Design sem avaliacao etica leva a dark patterns eventualmente"
      instead: "Avaliar CADA nova mecanica na Matrix antes de implementar"
